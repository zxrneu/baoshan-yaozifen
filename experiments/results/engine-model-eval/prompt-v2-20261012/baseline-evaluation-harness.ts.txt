import fs from 'node:fs';
import { LlmPromptBuilder as Legacy } from './.prompt-legacy';
import { LlmPromptBuilder } from './src/ai/engines/llm/llmPrompts';
import { LlmEngine } from './src/ai/engines/llmEngine';
import { BenchmarkRunner } from './src/ai/eval/benchmarkRunner';
import { ALL_BENCHMARK_SCENARIOS } from './src/ai/eval/scenarios';
import { contentHash } from './src/ai/eval/fingerprint';
import { evaluationSourceHash, saveEvaluationRun } from './scripts/eval-artifacts';
// 当前传输/裁判保持一致，只替换独立加载的历史提示词构造器。
LlmPromptBuilder.buildBiddingMessages = Legacy.buildBiddingMessages;
LlmPromptBuilder.buildTrumpMessages = Legacy.buildTrumpMessages;
LlmPromptBuilder.buildBuryMessages = Legacy.buildBuryMessages;
LlmPromptBuilder.buildPlayMessages = Legacy.buildPlayMessages;
const engine = new LlmEngine({ model: process.env.LLM_MODEL ?? 'qwen38_27b',
 baseUrl: process.env.LLM_BASE_URL, apiKey: process.env.LLM_API_KEY, protocol: 'chat',
 responseFormat: 'json_object', reasoningEffort: 'none', timeoutMs: 30000,
 enableDevMockOnFailure: false, maxConcurrency: 1 });
const runner = new BenchmarkRunner();
const summary = await runner.runSuite(engine, ALL_BENCHMARK_SCENARIOS, {
 runId: `legacy-${Date.now()}-qwen38_27b`, modelName: process.env.LLM_MODEL ?? 'qwen38_27b',
 concurrency: 1, codeHash: await evaluationSourceHash(), engineConfig: {
  promptVariant: 'c2819c8',
  promptSourceHash: await contentHash(fs.readFileSync('/private/tmp/baoshan-original-prompt-source.txt', 'utf8')),
 },
});
await saveEvaluationRun(runner, summary, '/private/tmp/baoshan-prompt-legacy-full');
console.log(JSON.stringify({ runId: summary.runId, passedCount: summary.passedCount,
 fallbackCount: summary.fallbackCount, macro: summary.phaseMacroAverageScore }));
